<product_spec>
<one_line>Mobile-first cocktail pantry web app (vanilla HTML/CSS/JS) — add ingredients to local pantry and show cocktails you can make using TheCocktailDB.</one_line>

<ui_requirements>
<pantry_view>
<description>Landing page shows pantry items or empty state text. Persistent bottom bar with Add (left), Drinks (center), Trash (right).</description>
</pantry_view>

<add_overlay>
<description>Slides in from right over pantry. Top bar: back icon, search input (real-time), three-dots menu (filters). Results show ingredient_tab cards. Clicking card slides ingredient_detail overlay.</description>
</add_overlay>

<ingredient_detail_overlay>
<description>Slides in from right over add_overlay. Shows large image, name, short description and big Add button bottom-center. Back returns to add_overlay.</description>
</ingredient_detail_overlay>

<drinks_list_page>
<description>Top bar: back, search, three-dots menu. Displays drink_tab cards sorted by match_score by default. Cards show image left, text right; background fade maps to match_score.</description>
</drinks_list_page>

<drink_detail_overlay>
<description>Slides in from right. Large image, name, short history, difficulty stars (default 3), prep time (default 2–3 mins), ingredient list with missing highlighted in red, instructions. Back returns to drinks list.</description>
</drink_detail_overlay>
</ui_requirements>

<data_integration>
<api>TheCocktailDB</api>
<endpoints>
<endpoint name="search_by_name">https://www.thecocktaildb.com/api/json/v1/1/search.php?s={name}</endpoint>
<endpoint name="lookup_by_id">https://www.thecocktaildb.com/api/json/v1/1/lookup.php?i={id}</endpoint>
<endpoint name="filter_by_ingredient">https://www.thecocktaildb.com/api/json/v1/1/filter.php?i={ingredient}</endpoint>
<endpoint name="list_all_ingredients">https://www.thecocktaildb.com/api/json/v1/1/list.php?i=list</endpoint>
</endpoints>
<fields_used>idDrink, strDrink, strInstructions, strDrinkThumb, strIngredient1..15, strMeasure1..15, strCategory, strAlcoholic</fields_used>
</data_integration>

<matching_algorithm>
<brief>
Normalize pantry items and drink ingredients to lowercase trimmed strings. For each fetched drink compute:
match_count, total_ingredients, missing_count, match_score = match_count / total_ingredients.
Sort drinks by match_score descending by default. Map match_score (0..1) to card background fade alpha (alpha = 0.15 + match_score*0.65).
</brief>
<debounce>150ms for searches</debounce>
<local_storage_key>pantry_items_v1</local_storage_key>
</matching_algorithm>
</product_spec>
